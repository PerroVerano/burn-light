cmake_minimum_required(VERSION 3.20)
project(appendcd VERSION 0.1.0 LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "This PoC targets native macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBISOFS REQUIRED IMPORTED_TARGET libisofs-1>=1.5.4)
pkg_check_modules(LIBUDFREAD REQUIRED IMPORTED_TARGET libudfread>=1.2.0)
find_program(CDRECORD_PATH appendcd-cdrecord PATHS "$ENV{HOME}/.local/libexec/appendcd" "${CMAKE_SOURCE_DIR}/build/backend" NO_DEFAULT_PATH REQUIRED)
execute_process(COMMAND "${CDRECORD_PATH}" -version OUTPUT_VARIABLE BACKEND_VERSION RESULT_VARIABLE BACKEND_STATUS)
if(NOT BACKEND_STATUS EQUAL 0 OR NOT BACKEND_VERSION MATCHES "3.02a09-appendcd-media6")
  message(FATAL_ERROR "Use the isolated appendcd media6 backend; run scripts/build-cdrecord.py first.")
endif()
find_program(READCD_PATH readcd REQUIRED)
find_program(INITIAL_CDRECORD_PATH appendcd-initial-cdrecord PATHS "$ENV{HOME}/.local/libexec/appendcd" "${CMAKE_SOURCE_DIR}/build/initial-backend" NO_DEFAULT_PATH REQUIRED)
execute_process(COMMAND "${INITIAL_CDRECORD_PATH}" -version OUTPUT_VARIABLE INITIAL_BACKEND_VERSION RESULT_VARIABLE INITIAL_BACKEND_STATUS)
if(NOT INITIAL_BACKEND_STATUS EQUAL 0 OR NOT INITIAL_BACKEND_VERSION MATCHES "3.02a09-appendcd-initial7")
  message(FATAL_ERROR "First-session support requires the separate initial7 backend; build with --initial --output-dir build/initial-backend.")
endif()
add_library(appendcore src/Util.cpp src/IsoCore.cpp src/IsoInitial.cpp src/IsoVerifier.cpp src/GeneratedMetadata.cpp src/DiscScanner.cpp src/OpticalDrive.cpp src/DiskArbitrationGuard.cpp src/UdfReader.cpp src/UdfSession.cpp)
target_link_libraries(appendcore PUBLIC PkgConfig::LIBISOFS PkgConfig::LIBUDFREAD "-framework CoreFoundation" "-framework IOKit")
target_include_directories(appendcore PUBLIC src)
target_compile_definitions(appendcore PUBLIC LIBISOFS_WITHOUT_LIBBURN CDRECORD_PATH="${CDRECORD_PATH}" INITIAL_CDRECORD_PATH="${INITIAL_CDRECORD_PATH}" READCD_PATH="${READCD_PATH}")
target_compile_options(appendcore PRIVATE -Wall -Wextra -Wpedantic -Wno-deprecated-declarations)
add_executable(appendcd src/main.cpp)
target_link_libraries(appendcd PRIVATE appendcore)
add_executable(guard_recovery_test tests/guard_recovery.cpp)
target_link_libraries(guard_recovery_test PRIVATE appendcore)
install(TARGETS appendcd RUNTIME DESTINATION bin)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME offline_multisession COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/offline.py $<TARGET_FILE:appendcd>)
add_test(NAME udf_multisession COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/udf_offline.py $<TARGET_FILE:appendcd>)
add_executable(backend_target_test tests/backend_target.cpp)
add_test(NAME backend_target COMMAND backend_target_test)
add_executable(recovery_image_test tests/recovery_image.cpp)
target_link_libraries(recovery_image_test PRIVATE appendcore)
add_test(NAME recovery_image COMMAND recovery_image_test "${CMAKE_SOURCE_DIR}/artifacts/udf-tests")
set_tests_properties(recovery_image PROPERTIES DEPENDS udf_multisession)
add_executable(drive_selection_test tests/drive_selection.cpp)
target_link_libraries(drive_selection_test PRIVATE appendcore)
add_test(NAME drive_selection COMMAND drive_selection_test)
add_test(NAME initial_session COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/initial.py $<TARGET_FILE:appendcd>)
